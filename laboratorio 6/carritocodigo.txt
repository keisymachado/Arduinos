// --- Motor Shield L298P ---
// Motor A conectado a M1 (izquierdo)
// Motor B conectado a M2 (derecho)

// Pines del Motor A
#define IN1 12 // Dirección motor izquierdo
#define IN2 13 // Dirección motor izquierdo
#define ENA 10 // Velocidad motor izquierdo

// Pines del Motor B
#define IN3 8 // Dirección motor derecho
#define IN4 9 // Dirección motor derecho
#define ENB 11 // Velocidad motor derecho

// Pines del receptor KRF100
#define D0 2  // Botón adelante
#define D1 3  // Botón atrás
#define D2 4  // Botón izquierda
#define D3 5  // Botón derecha

// Compensación de potencia para los motores
#define COMP_IZQ 10
#define COMP_DER 10

// Velocidad base de los motores
#define VELOCIDAD_BASE 240

// Sensor ultrasónico HC-SR04
#define TRIG_PIN A0 // Pin Trigger
#define ECHO_PIN A1 // Pin Echo
#define DISTANCIA_MINIMA 20  // Distancia mínima permitida en cm

void setup() {
  Serial.begin(9600); // Inicia comunicación serial

  // Configuración de pines del motor
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(ENA, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  pinMode(ENB, OUTPUT);

  // Configuración de pines del receptor
  pinMode(D0, INPUT);
  pinMode(D1, INPUT);
  pinMode(D2, INPUT);
  pinMode(D3, INPUT);

  // Configuración de pines del sensor ultrasónico
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  Serial.println("Sistema listo...");
}

// Función para medir distancia con el HC-SR04
long medirDistancia() {

  // Genera pulso ultrasónico
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);

  // Lee el tiempo del eco
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);

  // Convierte tiempo en distancia
  long distancia = duration * 0.034 / 2;

  // Si falla la lectura, asume distancia libre
  if (distancia == 0)
    distancia = 400;

  return distancia;
}

void loop() {

  // Lectura de botones del control remoto
  bool adelante = digitalRead(D0);
  bool atras = digitalRead(D1);
  bool izquierda = digitalRead(D2);
  bool derecha = digitalRead(D3);

  // Si se presiona avanzar
  if (adelante) {

    // Mide distancia al frente
    long distancia = medirDistancia();

    Serial.print("Distancia: ");
    Serial.print(distancia);
    Serial.println(" cm");

    // Si no hay obstáculo
    if (distancia > DISTANCIA_MINIMA) {
      Adelante();
    }

    // Si hay obstáculo cerca
    else {
      Parar();
      Serial.println("⚠️ Obstáculo detectado — bloqueado");
    }
  }

  // Movimiento hacia atrás
  else if (atras) {
    Atras();
  }

  // Giro a la izquierda
  else if (izquierda) {
    Izquierda();
  }

  // Giro a la derecha
  else if (derecha) {
    Derecha();
  }

  // Si no se presiona ningún botón
  else {
    Parar();
  }
}

// --- FUNCIONES DE MOVIMIENTO ---

// Movimiento hacia adelante
void Adelante() {

  Serial.println("→ Adelante");

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  analogWrite(ENA, limitar(VELOCIDAD_BASE + COMP_IZQ));

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  analogWrite(ENB, limitar(VELOCIDAD_BASE + COMP_DER));
}

// Movimiento hacia atrás
void Atras() {

  Serial.println("← Atrás");

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  analogWrite(ENA, limitar(VELOCIDAD_BASE + COMP_IZQ));

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  analogWrite(ENB, limitar(VELOCIDAD_BASE + COMP_DER));
}

// Giro a la izquierda
void Izquierda() {

  Serial.println("↺ Izquierda");

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, HIGH);
  analogWrite(ENA, 0);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  analogWrite(ENB, limitar(VELOCIDAD_BASE + COMP_DER));
}

// Giro a la derecha
void Derecha() {

  Serial.println("↻ Derecha");

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  analogWrite(ENA, limitar(VELOCIDAD_BASE + COMP_IZQ));

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, HIGH);
  analogWrite(ENB, 0);
}

// Detiene los motores
void Parar() {

  Serial.println("■ Parado");

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, HIGH);

  analogWrite(ENA, 0);
  analogWrite(ENB, 0);
}

// Limita valores entre 0 y 255
int limitar(int valor) {

  if (valor > 255)
    return 255;

  if (valor < 0)
    return 0;

  return valor;
}
